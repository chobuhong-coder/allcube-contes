// ===== 매주 월요일 05:00(KST) 자동 실행: 스크램블 5개 생성 후 Firebase 등록 ===== // Cron Trigger: 0 20 * * 0 (UTC 기준 일요일 20:00 = 한국시간 월요일 05:00) const DB_URL = 'https://allcube-booking-default-rtdb.firebaseio.com'; const FACES = ['U', 'D', 'L', 'R', 'F', 'B']; const AXIS = { U: 'UD', D: 'UD', L: 'LR', R: 'LR', F: 'FB', B: 'FB' }; const MODS = ['', "'", '2']; function randomScramble(length = 20) { const seq = []; let prevFace = null; let prevAxis = null; while (seq.length < length) { const face = FACES[Math.floor(Math.random() * FACES.length)]; const axis = AXIS[face]; if (face === prevFace) continue; if (axis === prevAxis) continue; // 같은 축(U/D, L/R, F/B) 연속 방지 const mod = MODS[Math.floor(Math.random() * MODS.length)]; seq.push(face + mod); prevAxis = axis; prevFace = face; } return seq.join(' '); } function pad(n) { return String(n).padStart(2, '0'); } // UTC 시각에 9시간을 더한 뒤 UTC getter로 읽으면 "한국시간 기준 필드"를 얻을 수 있음 function kstShifted(date) { return new Date(date.getTime() + 9 * 3600 * 1000); } function isoDateKST(d) { return `${d.getUTCFullYear()}-${pad(d.getUTCMonth() + 1)}-${pad(d.getUTCDate())}`; } function base64url(input) { let bytes; if (typeof input === 'string') bytes = new TextEncoder().encode(input); else bytes = new Uint8Array(input); let str = ''; bytes.forEach((b) => { str += String.fromCharCode(b); }); return btoa(str).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, ''); } function pemToArrayBuffer(pem) { const b64 = pem.replace(/-----BEGIN PRIVATE KEY-----/, '') .replace(/-----END PRIVATE KEY-----/, '') .replace(/\s/g, ''); const binary = atob(b64); const bytes = new Uint8Array(binary.length); for (let i = 0; i < binary.length; i++) bytes[i] = binary.charCodeAt(i); return bytes.buffer; } async function getGoogleAccessToken(saJson) { const sa = JSON.parse(saJson); const now = Math.floor(Date.now() / 1000); const header = { alg: 'RS256', typ: 'JWT' }; const claims = { iss: sa.client_email, scope: 'https://www.googleapis.com/auth/firebase.database https://www.googleapis.com/auth/userinfo.email', aud: 'https://oauth2.googleapis.com/token', iat: now, exp: now + 3600 }; const unsigned = `${base64url(JSON.stringify(header))}.${base64url(JSON.stringify(claims))}`; const keyData = pemToArrayBuffer(sa.private_key); const cryptoKey = await crypto.subtle.importKey( 'pkcs8', keyData, { name: 'RSASSA-PKCS1-v1_5', hash: 'SHA-256' }, false, ['sign'] ); const signature = await crypto.subtle.sign( 'RSASSA-PKCS1-v1_5', cryptoKey, new TextEncoder().encode(unsigned) ); const jwt = `${unsigned}.${base64url(signature)}`; const res = await fetch('https://oauth2.googleapis.com/token', { method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded' }, body: `grant_type=urn:ietf:params:oauth:grant-type:jwt-bearer&assertion=${jwt}` }); const data = await res.json(); if (!data.access_token) throw new Error('토큰 발급 실패: ' + JSON.stringify(data)); return data.access_token; } async function generateAndPublish(env) { const monday = kstShifted(new Date()); // 트리거 시점이 이미 월요일 05시(KST) const nextMonday = new Date(monday.getTime() + 7 * 86400000); const weekLabel = isoDateKST(monday); const startAt = `${isoDateKST(monday)}T05:00:00+09:00`; const endAt = `${isoDateKST(nextMonday)}T05:00:00+09:00`; const payload = { weekLabel, event: '333', eventName: '3x3x3', startAt, endAt, scramble1: randomScramble(), scramble2: randomScramble(), scramble3: randomScramble(), scramble4: randomScramble(), scramble5: randomScramble() }; const accessToken = await getGoogleAccessToken(env.FIREBASE_SA_KEY); const res = await fetch(`${DB_URL}/contests/current.json`, { method: 'PUT', headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${accessToken}` }, body: JSON.stringify(payload) }); if (!res.ok) { const text = await res.text(); throw new Error(`Firebase 쓰기 실패 (${res.status}): ${text}`); } return payload; } export default { async scheduled(event, env, ctx) { ctx.waitUntil(generateAndPublish(env)); }, // 대시보드에서 수동 테스트용 (Cron Triggers 탭의 "Trigger" 버튼으로도 테스트 가능) async fetch(request, env) { try { const result = await generateAndPublish(env); return new Response(JSON.stringify(result, null, 2), { headers: { 'Content-Type': 'application/json' } }); } catch (e) { return new Response('오류: ' + e.message, { status: 500 }); } } };
